// Scoring Engine & Pipeline Execution (길목 보완 안전장치 추가) async function calculateAndRender() { if (!currentPos) return; const logEl = document.getElementById('pipeline-log'); logEl.innerText = '실시간 기상 및 POI 스코어 계산 중...'; const radius = currentMode === 1 ? 5000 : 10000; // Parallel Fetching: Overpass POI + Open-Meteo Weather const [overpassData, weatherData] = await Promise.all([ fetchOverpassPOIs(currentPos.lat, currentPos.lng, radius), fetchRealtimeWeather(currentPos.lat, currentPos.lng) ]); let pool = [...overpassData, ...PRESET_HOTSPOTS]; if (currentMode === 1) { pool = pool.filter(p => isInsideWisu(p.lat, p.lng)); } // Determine FINAL Target let finalTarget = null; if (currentMode === 1) { let minD = Infinity; MAJOR_HUBS.forEach(h => { const d = getDistanceKm(currentPos.lat, currentPos.lng, h.lat, h.lng); if (d < minD) { minD = d; finalTarget = { ...h, dist: d }; } }); } else { const wisuPool = pool.filter(p => isInsideWisu(p.lat, p.lng)); let minD = Infinity; wisuPool.forEach(p => { const d = getDistanceKm(currentPos.lat, currentPos.lng, p.lat, p.lng); if (d < minD) { minD = d; finalTarget = { ...p, dist: d }; } }); if (!finalTarget) { finalTarget = { ...MAJOR_HUBS[0], dist: getDistanceKm(currentPos.lat, currentPos.lng, MAJOR_HUBS[0].lat, MAJOR_HUBS[0].lng) }; } } // Integrated Scoring Engine const nowHour = new Date().getHours(); const finalBearing = getBearing(currentPos.lat, currentPos.lng, finalTarget.lat, finalTarget.lng); const scoredPool = pool.map(item => { const dist = getDistanceKm(currentPos.lat, currentPos.lng, item.lat, item.lng); let score = 50; score += getDynamicTimeCategoryBonus(item.amenity, item.shop, nowHour); if (weatherData.isRainOrSnow) score += 20; const logResult = calculateCallLogScore(item.lat, item.lng); score += logResult.bonus; const itemBearing = getBearing(currentPos.lat, currentPos.lng, item.lat, item.lng); const delta = Math.abs(finalBearing - itemBearing); if (delta <= 45) score += 30; if (delta >= 120) score -= 50; score -= (dist * 4); return { ...item, dist, score: Math.round(score), delta, hasWeatherBonus: weatherData.isRainOrSnow, hasLogBonus: logResult.matched }; }); // Separation & Filtering Logic const minSeparation = currentMode === 1 ? 2.0 : 0.5; let selectedRoute = []; scoredPool.sort((a, b) => b.score - a.score); for (let candidate of scoredPool) { if (candidate.dist >= finalTarget.dist * 0.95) continue; if (getDistanceKm(candidate.lat, candidate.lng, finalTarget.lat, finalTarget.lng) < 0.2) continue; const isSeparated = selectedRoute.every(s => getDistanceKm(candidate.lat, candidate.lng, s.lat, s.lng) >= minSeparation); if (isSeparated) selectedRoute.push(candidate); if (selectedRoute.length === 3) break; } // [보완 안전장치] 길목 후보가 3개 미만일 때 경로 보간(Interpolation)으로 주요 거점 생성 if (selectedRoute.length < 3 && currentMode === 2) { const needed = 3 - selectedRoute.length; const ratios = [0.25, 0.5, 0.75]; const fallbackNames = ["복귀 길목 거점 A", "복귀 길목 거점 B", "복귀 길목 거점 C"]; for (let i = 0; i < needed; i++) { const ratio = ratios[i % ratios.length]; const interLat = currentPos.lat + (finalTarget.lat - currentPos.lat) * ratio; const interLng = currentPos.lng + (finalTarget.lng - currentPos.lng) * ratio; const interDist = getDistanceKm(currentPos.lat, currentPos.lng, interLat, interLng); selectedRoute.push({ name: `${finalTarget.name.split(' ')[0]} 방향 ${fallbackNames[i]}`, lat: interLat, lng: interLng, dist: interDist, score: 70 - i * 5, delta: 0, isWisu: false, address: "귀가 주요 이동 경로" }); } } selectedRoute.sort((a, b) => a.dist - b.dist); const weatherText = weatherData.isRainOrSnow ? ' · 비/눈(+20점)' : ''; logEl.innerText = `방금 · ${radius/1000}km 이내 · ${scoredPool.length}곳 검출${weatherText} · FINAL: ${finalTarget.name}`; renderCards(selectedRoute, finalTarget); }
